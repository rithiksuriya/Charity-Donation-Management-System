<script>
  document.getElementById('year').textContent = new Date().getFullYear();
  const donors = [];

  function submitDonation(e) {
    e.preventDefault();
    const name = document.getElementById('name').value.trim();
    const email = document.getElementById('email').value.trim();
    const amount = parseFloat(document.getElementById('amount').value);
    const message = document.getElementById('message').value.trim() || 'N/A';

    if (!name || !email || amount <= 0) {
      alert("Please fill all required fields with valid data.");
      return;
    }

    donors.push({ name, email, amount, message });
    updateDonorList();
    alert("Thank you for your donation!");

    showSlide('detailsSlide');
    document.getElementById('donationForm').reset();
  }

  function updateDonorList() {
    const donorList = document.getElementById('donorList');
    donorList.innerHTML = '';
    donors.forEach((donor, index) => {
      const li = document.createElement('li');
      li.innerHTML = `<strong>Donor #${index + 1}</strong><br>
        Name: ${donor.name}<br>Email: ${donor.email}<br>
        Amount: $${donor.amount.toFixed(2)}<br>Message: ${donor.message}`;
      donorList.appendChild(li);
    });
  }

  function showSlide(id) {
    const slides = document.querySelectorAll('.slide');
    slides.forEach(slide => slide.classList.remove('active'));
    const target = document.getElementById(id);
    if (target) {
      target.classList.add('active');
    }
    if (id === 'summarySlide') updateSummary();
    if (id === 'advancedSlide') clearFilterResults();
  }

  function goBack() {
    showSlide('formSlide');
  }

  function goToSummary() {
    showSlide('summarySlide');
  }

  function goToAdvanced() {
    showSlide('advancedSlide');
  }

  function goBackToSummary() {
    showSlide('summarySlide');
  }

  function updateSummary() {
    document.getElementById('totalDonors').textContent = donors.length;
    const total = donors.reduce((acc, donor) => acc + donor.amount, 0);
    document.getElementById('totalAmount').textContent = total.toFixed(2);
  }

  function downloadDonorsAsPDF() {
    if (donors.length === 0) {
      alert("No donors to export.");
      return;
    }
    const { jsPDF } = window.jspdf || window.jsPDF;
    const doc = new jsPDF();
    doc.setFontSize(14);
    doc.text("Donor List", 14, 15);
    donors.forEach((donor, i) => {
      let y = 25 + i * 10;
      if (y > 280) {
        doc.addPage();
        y = 15;
      }
      doc.text(`${i + 1}. ${donor.name} - $${donor.amount.toFixed(2)}`, 14, y);
    });
    doc.save("donor-list.pdf");
  }

  function filterDonors() {
    const minAmountInput = document.getElementById('filterAmount').value;
    const minAmount = parseFloat(minAmountInput);
    const filteredListEl = document.getElementById('filteredList');
    filteredListEl.innerHTML = "";

    if (isNaN(minAmount) || minAmount < 0) {
      alert("Please enter a valid minimum amount.");
      return;
    }

    const filtered = donors.filter(donor => donor.amount >= minAmount);
    if (filtered.length === 0) {
      filteredListEl.textContent = `No donors found with amount >= $${minAmount.toFixed(2)}.`;
      return;
    }

    filtered.forEach(donor => {
      const div = document.createElement('div');
      div.textContent = `${donor.name} donated $${donor.amount.toFixed(2)}`;
      filteredListEl.appendChild(div);
    });
  }

  function clearFilterResults() {
    document.getElementById('filterAmount').value = "";
    document.getElementById('filteredList').innerHTML = "";
  }
</script>
